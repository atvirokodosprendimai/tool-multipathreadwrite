"""Post-hoc offline accounting for the two saved synthetic pilots; no model calls.

Usage: python3 trace-diagnosis.py.txt GROUPED_RAW_ROOT SMOKE_RAW_ROOT PUBLIC_REPORT_DIR
Raw transcripts stay private. Output contains sizes, hashes and counts, never their text.
"""
from collections import Counter
from decimal import Decimal
import hashlib
import json
from pathlib import Path
import sys

KEYS = ("input_tokens", "output_tokens", "cache_read_input_tokens", "cache_creation_input_tokens")
# Empirical accounting coefficients; validated against every receipt, not a tariff claim.
RATES = dict(zip(KEYS, map(Decimal, ("2", "10", "0.1", "4"))))
PHASES = {
    "grouped-structural.A": ["inspect", "inspect", "edit_and_check"],
    "grouped-structural.B": ["inspect", "edit_and_check"],
    "grouped-config.A": ["inspect", "edit_and_check"],
    "grouped-config.B": ["inspect", "inspect", "edit_and_check"],
    "grouped-source-tests.A": ["inspect", "edit_and_check"],
    "grouped-source-tests.B": ["inspect", "edit_and_check"],
    "grouped-docs.A": ["inspect", "edit_and_check"],
    "grouped-docs.B": ["inspect", "edit", "check"],
    "grouped-routing.A": ["inspect", "edit_and_check"],
    "grouped-routing.B": ["inspect", "edit_and_check"],
    "pilot-singleline.A": ["inspect", "inspect", "edit", "inspect", "check"],
    "pilot-singleline.B": ["inspect", "edit_and_check"],
    "pilot-multisite.A": ["inspect", "inspect", "edit", "check"],
    "pilot-multisite.B": ["inspect", "edit_and_check"],
}


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def components(usage):
    return {key: float(Decimal(usage[key]) * RATES[key] / 1000000) for key in KEYS}


def analyze(cell, path, expected_sha):
    raw = path.read_bytes()
    assert sha(raw) == expected_sha, "transcript fingerprint differs"
    rows = [json.loads(line) for line in raw.splitlines()]
    assert all(isinstance(row, dict) for row in rows)
    finals = [row for row in rows if row.get("type") == "result"]
    assert len(finals) == 1
    result = finals[0]
    assert result["usage"] == cell["record"]["usage"]
    assert result["total_cost_usd"] == cell["record"]["cost_usd"]
    usage = {key: result["usage"][key] for key in KEYS}
    assert all(type(value) is int and value >= 0 for value in usage.values())
    costs = components(usage)
    assert abs(sum(costs.values()) - result["total_cost_usd"]) < 1e-12
    messages, uses, answers = {}, {}, {}
    for row in rows:
        if row.get("type") == "assistant":
            message = row["message"]
            inputs = {key: message["usage"][key] for key in KEYS if key != "output_tokens"}
            if message["id"] not in messages:
                messages[message["id"]] = {
                    "inputs": inputs, "context_tokens": sum(inputs.values()),
                    "frame_output_tokens": message["usage"]["output_tokens"], "text_chars": 0,
                }
            assert messages[message["id"]]["inputs"] == inputs
            for part in message["content"]:
                if part["type"] == "tool_use":
                    assert part["id"] not in uses
                    uses[part["id"]] = part
                elif part["type"] == "text":
                    messages[message["id"]]["text_chars"] += len(part["text"])
        elif row.get("type") == "user":
            for part in row.get("message", {}).get("content", []):
                if part.get("type") == "tool_result":
                    assert part["tool_use_id"] not in answers
                    answers[part["tool_use_id"]] = part
    assert set(uses) == set(answers)
    # The CLI turn counter is not always the count of emitted model messages.
    assert type(result["num_turns"]) is int and result["num_turns"] > 0
    for key in usage:
        if key != "output_tokens":
            assert sum(message["inputs"][key] for message in messages.values()) == usage[key]
    identity = cell["task"] + "." + cell["arm"]
    assert len(PHASES[identity]) == len(uses)
    calls = []
    for (tool_id, part), phase in zip(uses.items(), PHASES[identity], strict=True):
        content = answers[tool_id]["content"]
        if not isinstance(content, str):
            # Some native tools return arrays of text blocks; count delivered text only.
            assert all(block["type"] == "text" for block in content)
            content = "".join(block["text"] for block in content)
        arguments = json.dumps(part["input"], ensure_ascii=False, sort_keys=True)
        command = part["input"].get("command", "")
        calls.append({"name":part["name"], "phase":phase, "argument_chars":len(arguments),
                      "command_chars":len(command), "result_chars":len(content),
                      "arguments_sha256":sha(arguments.encode()), "result_sha256":sha(content.encode()),
                      "is_error":answers[tool_id].get("is_error"),
                      "mrw_write_mention":"mrw write" in command,
                      "mrw_instructions_mention":"mrw instructions" in command,
                      "declared_check_pass_mentions":content.count("check PASS (exit 0,")})
    return {"task":cell["task"], "arm":cell["arm"], "transcript_sha256":expected_sha,
            "solved":cell.get("solved", cell.get("verdict", {}).get("solved")),
            "turns":result["num_turns"], "usage":usage, "cost_usd":result["total_cost_usd"],
            "cost_components_usd":costs, "requests":list(messages.values()), "tool_calls":calls,
            "permission_denial_count":len(result["permission_denials"]),
            "stream_frame_output_sum":sum(message["frame_output_tokens"] for message in messages.values())}


def aggregate(cells, arm):
    chosen = [cell for cell in cells if cell["arm"] == arm]
    calls = [call for cell in chosen for call in cell["tool_calls"]]
    usage = {key:sum(cell["usage"][key] for cell in chosen) for key in KEYS}
    return {"turns":sum(cell["turns"] for cell in chosen), "tool_calls":len(calls), "usage":usage,
            "cost_usd":sum(cell["cost_usd"] for cell in chosen), "cost_components_usd":components(usage),
            "emitted_messages":sum(len(cell["requests"]) for cell in chosen),
            "first_context_tokens":[cell["requests"][0]["context_tokens"] for cell in chosen],
            "tool_kinds":dict(Counter(call["name"] for call in calls)),
            "command_chars":sum(call["command_chars"] for call in calls),
            "result_chars":sum(call["result_chars"] for call in calls),
            "results_by_phase":{phase:sum(call["result_chars"] for call in calls if call["phase"] == phase)
                                for phase in sorted({call["phase"] for call in calls})},
            "explicit_tool_errors":sum(call["is_error"] is True for call in calls),
            "permission_denials":sum(cell["permission_denial_count"] for cell in chosen),
            "mrw_write_mentions":sum(call["mrw_write_mention"] for call in calls),
            "instructions_command_mentions":sum(call["mrw_instructions_mention"] for call in calls)}


def main():
    grouped_root, smoke_root, public = map(Path, sys.argv[1:])
    output = {"kind":"post-hoc offline trace diagnosis", "analyst_sha256":sha(Path(__file__).read_bytes()),
              "scope":"14 task cells; setup costs stay in original reports, excluded here",
              "empirical_usd_per_million_tokens":{key:float(value) for key,value in RATES.items()},
              "rate_note":"Algebraically reconciles these receipts; not an independently verified current tariff.",
              "phase_note":"Manual labels for these observed commands; mixed commands are not cost-separated.",
              "output_note":"Final result output_tokens are authoritative; streamed frame counters are incomplete.",
              "experiments":{}}
    for name, filename, root in [("grouped","grouped-pilot.json",grouped_root),
                                  ("smoke","claude-pilot.json",smoke_root)]:
        source = (public / filename).read_bytes()
        cells = []
        for cell in json.loads(source)["cells"]:
            if name == "grouped":
                identity = cell["task"] + "." + cell["arm"]
                path = root / "transcripts" / (identity + ".jsonl")
                fingerprint = cell["transcript_sha256"]
            else:
                path = root / "attempts" / cell["task"] / ("canary." + cell["arm"] + ".transcript.jsonl")
                fingerprint = cell["raw_transcript_sha256"]
            cells.append(analyze(cell, path, fingerprint))
        assert len(cells) == (10 if name == "grouped" else 4)
        assert all(cell["solved"] for cell in cells)
        totals = {arm:aggregate(cells,arm) for arm in "AB"}
        deltas = {key:totals["B"]["cost_components_usd"][key] - totals["A"]["cost_components_usd"][key]
                  for key in KEYS}
        output["experiments"][name] = {"source_sha256":sha(source), "cells":cells, "totals":totals,
                                       "B_minus_A_cost_components_usd":deltas}
    print(json.dumps(output, ensure_ascii=False, indent=2))


if __name__ == "__main__":
    main()
